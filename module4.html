1. Multi-Stage Pipeline (`ci-cd-pipeline.yml`) [AB]

Automated GitHub Actions workflow performing static security scans, unit tests, and image generation.

name: Garang Guni Continuous Integration

on:
  push:
    branches: [ main, develop ]

jobs:
  build-and-test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - name: Set up JDK 17
        uses: actions/setup-java@v3
        with:
          java-version: '17'
          distribution: 'temurin'

      - name: Security Vulnerability Scan (Snyk)
        uses: snyk/actions/maven@master
        env:
          SNYK_TOKEN: ${{ secrets.SNYK_TOKEN }}

      - name: Run Static Analysis & Unit Tests
        run: mvn compile spotbugs:check test
